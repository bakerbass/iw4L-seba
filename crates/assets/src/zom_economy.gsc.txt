// Zombies economy driven by the map's own entities:
//   zombie_door         trigger_use -> script_brushmodel targets (zombie_cost)
//   weapon_upgrade      trigger_use wall buys (zombie_weapon_upgrade)
//   treasure_chest_use  trigger_use mystery box locations (zombie_cost)
//   script_brushmodel   script_parameters "board" groups sharing a targetname: window barricades
// Guns come from a gun set so the BO weapon names can be swapped for MW2 ones.

main()
{
	waittillframeend;
	level.zom_gunset = getDvar("zom_gunset");
	if (level.zom_gunset == "")
		level.zom_gunset = "mw2";
	level.zom_start_points = 500;
	level.zom_box_cost = 950;
	thread onConnect();
	doors = getEntArray("zombie_door", "targetname");
	for (i = 0; i < doors.size; i++)
		doors[i] thread doorThink();
	buys = getEntArray("weapon_upgrade", "targetname");
	for (i = 0; i < buys.size; i++)
		buys[i] thread wallBuyThink();
	thread boxSetup();
	thread barricadeSetup();
	papMachines = getEntArray("zombie_vending_upgrade", "targetname");
	for (i = 0; i < papMachines.size; i++)
		papMachines[i] thread papThink();
	machines = getEntArray("zombie_vending", "targetname");
	for (i = 0; i < machines.size; i++)
		machines[i] thread perkThink();
}

onConnect()
{
	for (;;)
	{
		for (i = 0; i < level.players.size; i++)
		{
			p = level.players[i];
			if (isDefined(p) && !isDefined(p.zom_init))
			{
				p.zom_init = true;
				p thread initPlayer();
			}
		}
		wait 0.25;
	}
}

initPlayer()
{
	self endon("disconnect");
	self.zom_points = level.zom_start_points;
	self.zom_perks = [];
	hud = newClientHudElem(self);
	hud.horzAlign = "left";
	hud.vertAlign = "bottom";
	hud.alignX = "left";
	hud.alignY = "bottom";
	hud.x = 24;
	hud.y = -110;
	hud.fontScale = 2;
	hud.hideWhenInMenu = true;
	self.zom_hud = hud;
	self updatePointsHud();
	self thread barricadeWatch();
	for (;;)
	{
		// Console aid until zombies award points: set zom_givepoints <n>.
		give = getDvarInt("zom_givepoints");
		if (give != 0)
		{
			self addPoints(give);
			setDvar("zom_givepoints", 0);
		}
		wait 0.25;
	}
}

addPoints(amount)
{
	self.zom_points += amount;
	self updatePointsHud();
}

updatePointsHud()
{
	if (isDefined(self.zom_hud))
		self.zom_hud setText("Points: " + self.zom_points);
}

cost_of(ent, fallback)
{
	if (isDefined(ent.zombie_cost))
		return int(ent.zombie_cost);
	return fallback;
}

setBuyHint(verb, price)
{
	self setCursorHint("HINT_NOICON");
	self setHintString("Hold ^3[{+activate}]^7 to " + verb + " [Cost: " + price + "]");
	self useTriggerRequireLookAt();
}

doorThink()
{
	price = cost_of(self, 1000);
	self setBuyHint("open door", price);
	// Doors and their clip brushes block the way until bought.
	slabs = getEntArray(self.target, "targetname");
	for (i = 0; i < slabs.size; i++)
		slabs[i] solid();
	for (;;)
	{
		self waittill("trigger", player);
		if (!isDefined(player.zom_points) || player.zom_points < price)
			continue;
		player addPoints(0 - price);
		self openDoor();
		return;
	}
}

openDoor()
{
	target = self.target;
	all = getEntArray("zombie_door", "targetname");
	for (i = 0; i < all.size; i++)
	{
		if (isDefined(all[i].target) && all[i].target == target && all[i] != self)
			all[i] delete();
	}
	parts = getEntArray(target, "targetname");
	for (i = 0; i < parts.size; i++)
		parts[i] thread slideOpen();
	self delete();
}

slideOpen()
{
	if (isDefined(self.script_noteworthy) && self.script_noteworthy == "clip")
	{
		self notsolid();
		self hide();
		return;
	}
	if (isDefined(self.script_angles))
	{
		self rotateTo(self.script_angles, 1.0, 0.1, 0.4);
		wait 1.0;
	}
	else if (isDefined(self.script_vector))
	{
		self moveTo(self.origin + self.script_vector, 1.0, 0.1, 0.4);
		wait 1.0;
	}
	else
	{
		self hide();
	}
	self notsolid();
}

// Window barricades. The map authors each window as a group of "board" brush models plus a
// clip brush, all sharing a targetname. Without zombies nothing tears boards off, so a share
// of them start torn (zom_barricade_damage percent, default 50) and players hold use to rebuild.
barricadeSetup()
{
	level.zom_barr = [];
	level.zom_board_points = 10;
	if (getDvar("zom_barricade_damage") == "")
		setDvar("zom_barricade_damage", 50);
	all = getEntArray();
	for (i = 0; i < all.size; i++)
	{
		e = all[i];
		if (!isDefined(e.script_parameters) || e.script_parameters != "board")
			continue;
		if (!isDefined(e.targetname))
			continue;
		g = undefined;
		for (j = 0; j < level.zom_barr.size; j++)
		{
			if (level.zom_barr[j].zom_name == e.targetname)
				g = level.zom_barr[j];
		}
		if (!isDefined(g))
		{
			g = spawn("script_origin", e.origin);
			g.zom_name = e.targetname;
			g.zom_boards = [];
			g.zom_sum = (0, 0, 0);
			level.zom_barr[level.zom_barr.size] = g;
		}
		g.zom_boards[g.zom_boards.size] = e;
		g.zom_sum = g.zom_sum + e.origin;
	}
	damage = getDvarInt("zom_barricade_damage");
	torn = 0;
	for (i = 0; i < level.zom_barr.size; i++)
	{
		g = level.zom_barr[i];
		g.zom_center = g.zom_sum * (1.0 / g.zom_boards.size);
		for (j = 0; j < g.zom_boards.size; j++)
		{
			g.zom_boards[j].zom_up = true;
			if (randomint(100) < damage)
			{
				g tearBoard(g.zom_boards[j]);
				torn++;
			}
		}
	}
	println("zombies: " + level.zom_barr.size + " barricades, " + torn + " boards torn");
}

tearBoard(board)
{
	board.zom_up = false;
	board hide();
	board notsolid();
}

mendBoard(board)
{
	board.zom_up = true;
	board show();
	board solid();
}

// First torn board of a barricade, or undefined when it is fully built.
tornBoard()
{
	for (i = 0; i < self.zom_boards.size; i++)
	{
		if (!self.zom_boards[i].zom_up)
			return self.zom_boards[i];
	}
	return undefined;
}

nearestTornBarricade(origin)
{
	best = undefined;
	bestDist = 110;
	for (i = 0; i < level.zom_barr.size; i++)
	{
		g = level.zom_barr[i];
		d = distance(origin, g.zom_center);
		if (d < bestDist && isDefined(g tornBoard()))
		{
			best = g;
			bestDist = d;
		}
	}
	return best;
}

barricadeWatch()
{
	self endon("disconnect");
	hint = newClientHudElem(self);
	hint.horzAlign = "center";
	hint.vertAlign = "middle";
	hint.alignX = "center";
	hint.alignY = "middle";
	hint.y = 60;
	hint.fontScale = 1.6;
	hint.hideWhenInMenu = true;
	held = 0;
	for (;;)
	{
		wait 0.1;
		if (!isDefined(level.zom_barr) || !isDefined(self.zom_points))
			continue;
		g = nearestTornBarricade(self.origin);
		if (!isDefined(g))
		{
			hint setText("");
			held = 0;
			continue;
		}
		hint setText("Hold ^3[{+activate}]^7 to rebuild the barricade");
		if (!self useButtonPressed())
		{
			held = 0;
			continue;
		}
		held += 0.1;
		if (held < 1.0)
			continue;
		held = 0;
		board = g tornBoard();
		if (isDefined(board))
		{
			g mendBoard(board);
			self addPoints(level.zom_board_points);
		}
	}
}

// Gun sets. BO names in the map data map onto the selected set.
gunFor(name)
{
	switch (name)
	{
	case "ithaca_zm":      return "spas12_mp";
	case "mp5k_zm":        return "mp5k_mp";
	case "m14_zm":         return "m21_mp";
	case "m16_zm":         return "m16_mp";
	case "ak74u_zm":       return "ak47_mp";
	case "mp40_zm":        return "ump45_mp";
	case "pm63_zm":        return "uzi_mp";
	case "rottweil72_zm":  return "model1887_mp";
	case "mpl_zm":         return "pp2000_mp";
	}
	return "";
}

priceFor(gun)
{
	switch (gun)
	{
	case "spas12_mp":
	case "model1887_mp":
		return 1200;
	case "m16_mp":
	case "ak47_mp":
		return 1000;
	case "m21_mp":
		return 500;
	}
	return 600;
}

wallBuyThink()
{
	gun = gunFor(self.zombie_weapon_upgrade);
	if (gun == "")
	{
		self delete();
		return;
	}
	price = priceFor(gun);
	self setBuyHint("buy weapon", price);
	for (;;)
	{
		self waittill("trigger", player);
		if (!isDefined(player.zom_points))
			continue;
		if (player hasWeapon(gun))
		{
			ammo = int(price / 2);
			if (player.zom_points < ammo)
				continue;
			player addPoints(0 - ammo);
			player giveMaxAmmo(gun);
			continue;
		}
		if (player.zom_points < price)
			continue;
		player addPoints(0 - price);
		player giveZombieGun(gun);
	}
}

giveZombieGun(gun)
{
	primaries = self getWeaponsListPrimaries();
	if (primaries.size >= 2)
		self takeWeapon(self getCurrentWeapon());
	self giveWeapon(gun);
	self giveMaxAmmo(gun);
	self switchToWeapon(gun);
}

boxSetup()
{
	boxes = getEntArray("treasure_chest_use", "targetname");
	if (boxes.size == 0)
		return;
	level.zom_boxes = boxes;
	level.zom_box_uses = 0;
	eligible = [];
	for (i = 0; i < boxes.size; i++)
	{
		if (!isDefined(boxes[i].start_exclude))
			eligible[eligible.size] = boxes[i];
	}
	if (eligible.size == 0)
		eligible = boxes;
	pick = eligible[randomint(eligible.size)];
	for (i = 0; i < boxes.size; i++)
	{
		boxes[i].zom_pool = boxPool();
		boxes[i] thread boxThink();
		if (boxes[i] != pick)
			boxes[i] boxOff();
	}
}

boxOn()
{
	self.zom_active = true;
	self makeUsable();
	self setBuyHint("search the box", level.zom_box_cost);
}

boxOff()
{
	self.zom_active = false;
	self makeUnusable();
}

boxThink()
{
	for (;;)
	{
		self waittill("trigger", player);
		if (!isDefined(self.zom_active) || !self.zom_active)
			continue;
		if (!isDefined(player.zom_points) || player.zom_points < level.zom_box_cost)
			continue;
		player addPoints(0 - level.zom_box_cost);
		gun = self.zom_pool[randomint(self.zom_pool.size)];
		player giveZombieGun(gun);
		level.zom_box_uses++;
		if (level.zom_box_uses >= 4 && randomint(6) == 0)
			level thread moveBox(self);
	}
}

moveBox(old)
{
	level.zom_box_uses = 0;
	old boxOff();
	others = [];
	for (i = 0; i < level.zom_boxes.size; i++)
	{
		if (level.zom_boxes[i] != old)
			others[others.size] = level.zom_boxes[i];
	}
	if (others.size == 0)
		return;
	others[randomint(others.size)] boxOn();
}

// Perks. Only the ones the runtime can express today are purchasable.
perkInfo(perk)
{
	info = [];
	switch (perk)
	{
	case "specialty_armorvest":
		info["name"] = "Juggernog";
		info["cost"] = 2500;
		info["ready"] = true;
		break;
	case "specialty_fastreload":
		info["name"] = "Speed Cola";
		info["cost"] = 3000;
		info["ready"] = true;
		break;
	case "specialty_rof":
		info["name"] = "Double Tap Root Beer";
		info["cost"] = 2000;
		info["ready"] = true;
		break;
	case "specialty_quickrevive":
		info["name"] = "Quick Revive";
		info["cost"] = 500;
		info["ready"] = false;
		break;
	default:
		info["name"] = perk;
		info["cost"] = 2000;
		info["ready"] = false;
	}
	return info;
}

perkThink()
{
	perk = self.script_noteworthy;
	info = perkInfo(perk);
	if (!info["ready"])
	{
		self setCursorHint("HINT_NOICON");
		self setHintString(info["name"] + " is not available yet");
		self useTriggerRequireLookAt();
		return;
	}
	self setBuyHint("drink " + info["name"], info["cost"]);
	for (;;)
	{
		self waittill("trigger", player);
		if (!isDefined(player.zom_points) || player.zom_points < info["cost"])
			continue;
		if (isDefined(player.zom_perks[perk]) && player.zom_perks[perk])
			continue;
		player addPoints(0 - info["cost"]);
		player.zom_perks[perk] = true;
		player applyPerk(perk);
	}
}

applyPerk(perk)
{
	if (perk == "specialty_armorvest")
	{
		self.maxhealth = 250;
		self.health = 250;
	}
	else if (perk == "specialty_fastreload")
	{
		self setPerk("specialty_fastreload");
	}
	else if (perk == "specialty_rof")
	{
		self setPerk("specialty_rof");
	}
}

// Perks are lost on death, as in Black Ops.
clearPerks()
{
	self.zom_perks = [];
}

// Pack-a-Punch: marks the gun with camo slot 5 (the sim doubles its damage)
// and adds the first attachment this gun supports.
papThink()
{
	price = 5000;
	self setBuyHint("pack-a-punch your weapon", price);
	for (;;)
	{
		self waittill("trigger", player);
		if (!isDefined(player.zom_points) || player.zom_points < price)
			continue;
		cur = player getCurrentWeapon();
		if (cur == "none" || cur == "")
			continue;
		if (!isDefined(player.zom_pap))
			player.zom_pap = [];
		if (isDefined(player.zom_pap[cur]) && player.zom_pap[cur])
			continue;
		player addPoints(0 - price);
		player thread papUpgrade(cur);
	}
}

papUpgrade(cur)
{
	self endon("disconnect");
	self endon("death");
	base = getSubStr(cur, 0, cur.size - 3);
	attachments = [];
	attachments[0] = "silencer";
	attachments[1] = "acog";
	attachments[2] = "reflex";
	attachments[3] = "fmj";
	attachments[4] = "xmags";
	attachments[5] = "grip";
	name = cur;
	for (i = 0; i < attachments.size; i++)
	{
		candidate = base + "_" + attachments[i] + "_mp";
		if (self weaponExists(candidate))
		{
			name = candidate;
			break;
		}
	}
	self takeWeapon(cur);
	wait 2.0;
	self giveWeapon(name, 5);
	self giveMaxAmmo(name);
	self switchToWeapon(name);
	self.zom_pap[name] = true;
}
