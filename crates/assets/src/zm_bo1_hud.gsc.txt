// BO1-style score and ammo readout for ZombiesT5. BO1 draws these from its HUD menu, which the
// engine cannot load; the values are the same ones the zombie scripts keep (score, weapon ammo).
#include common_scripts\utility;

main()
{
    level thread connected_flag();
    level thread close_windows();
    level thread unstick_zombies();
    for(;;)
    {
        level waittill("connecting", player);
        player thread bo1_hud();
    }
}

// BO1's map bakes every window clip as path-disconnected; here a window is only closed when a
// script says so, so close each barrier's clip once the blocker setup has run.
close_windows()
{
    while(!isdefined(level.exterior_goals))
    {
        wait 0.05;
    }
    wait 1;
    for(i = 0; i < level.exterior_goals.size; i++)
    {
        goal = level.exterior_goals[i];
        if(isdefined(goal.clip))
        {
            goal.clip disconnectpaths();
        }
    }
}

// A zombie whose entrance cannot be reached on foot stands still at its spawner. Send one that
// has not reached its entrance and has not moved for about four seconds to the next-closest
// window it has not tried yet.
unstick_zombies()
{
    for(;;)
    {
        wait 2;
        if(!isdefined(level.exterior_goals))
        {
            continue;
        }
        zs = getaiarray("axis");
        for(i = 0; i < zs.size; i++)
        {
            z = zs[i];
            if(!isdefined(z.first_node) || (isdefined(z.got_to_entrance) && z.got_to_entrance))
            {
                continue;
            }
            if(!isdefined(z.unstick_pos) || distance(z.origin, z.unstick_pos) >= 16)
            {
                z.unstick_pos = z.origin;
                z.unstick_waits = 0;
                continue;
            }
            z.unstick_waits++;
            if(z.unstick_waits < 2)
            {
                continue;
            }
            z.unstick_waits = 0;
            if(!isdefined(z.tried_goals))
            {
                z.tried_goals = [];
            }
            best = -1;
            best_dist = 99999999;
            for(j = 0; j < level.exterior_goals.size; j++)
            {
                skip = false;
                for(k = 0; k < z.tried_goals.size; k++)
                {
                    if(z.tried_goals[k] == j)
                    {
                        skip = true;
                    }
                }
                if(skip || level.exterior_goals[j] == z.first_node)
                {
                    continue;
                }
                d = distance(z.origin, level.exterior_goals[j].origin);
                if(d < best_dist)
                {
                    best_dist = d;
                    best = j;
                }
            }
            if(best < 0)
            {
                z.tried_goals = [];
                continue;
            }
            z.tried_goals[z.tried_goals.size] = best;
            z.first_node = level.exterior_goals[best];
            z setgoalpos(level.exterior_goals[best].origin);
        }
    }
}

// The engine raises this once every expected player is in; BO1's start score, intro and
// per-player setup all wait on it. Solo: the first player is everyone.
connected_flag()
{
    level waittill("connecting", player);
    while(!isdefined(level.flag) || !isdefined(level.flag["all_players_connected"]))
    {
        wait 0.05;
    }
    wait 0.5;
    flag_set("all_players_connected");
    wait 1;
    // BO1 solo starts with 500; its zombiemode.csv (which says so) is not available here.
    start = 500;
    if(getdvarint("zm_start_score") > 0)
    {
        start = getdvarint("zm_start_score");
    }
    if(player.score < start)
    {
        player.score = start;
        player.score_total = start;
        player.old_score = start;
    }
}

make_value(player, x, y, scale, r, g, b)
{
    elem = newclienthudelem(player);
    elem.horzalign = "user_right";
    elem.vertalign = "user_bottom";
    elem.alignx = "right";
    elem.aligny = "bottom";
    elem.x = x;
    elem.y = y;
    elem.fontscale = scale;
    elem.color = (r, g, b);
    elem.alpha = 1;
    elem.sort = 2;
    elem.foreground = true;
    elem.hidewheninmenu = false;
    elem.hidewhendead = true;
    return elem;
}

bo1_hud()
{
    self endon("disconnect");
    score = make_value(self, -30, -90, 2.2, 1, 0.85, 0.35);
    clip = make_value(self, -86, -40, 2.4, 1, 1, 1);
    slash = make_value(self, -76, -44, 1.6, 0.8, 0.8, 0.8);
    slash settext("/");
    stock = make_value(self, -30, -44, 1.6, 0.8, 0.8, 0.8);
    last_score = -1;
    last_clip = -1;
    last_stock = -1;
    for(;;)
    {
        if(isdefined(self.score) && self.score != last_score)
        {
            last_score = self.score;
            score setvalue(last_score);
        }
        weapon = self getcurrentweapon();
        if(isdefined(weapon) && weapon != "none")
        {
            c = self getweaponammoclip(weapon);
            s = self getammocount(weapon) - c;
            if(s < 0)
            {
                s = 0;
            }
            if(c != last_clip)
            {
                last_clip = c;
                clip setvalue(c);
            }
            if(s != last_stock)
            {
                last_stock = s;
                stock setvalue(s);
            }
        }
        wait 0.1;
    }
}
